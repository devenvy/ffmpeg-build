cmake_minimum_required(VERSION 3.20)
project(ffb_openvino_installed_sdk LANGUAGES CXX)
find_package(OpenVINO REQUIRED COMPONENTS Runtime CONFIG)
foreach(probe cxx bridges exception exception_static)
    if(probe STREQUAL "exception_static")
        set(source ../public-exception.cpp)
    else()
        set(source ../public-${probe}.cpp)
    endif()
    add_executable(${probe} ${source})
    set_target_properties(${probe} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
    target_compile_options(${probe} PRIVATE -UNDEBUG)
    target_link_libraries(${probe} PRIVATE openvino::runtime)
endforeach()
target_link_options(exception_static PRIVATE -static-libstdc++ -shared-libgcc)
