cmake_minimum_required(VERSION 3.20)
project(ffb_torch_installed_sdk LANGUAGES CXX)
find_package(Torch REQUIRED CONFIG)
find_package(Threads REQUIRED)
add_executable(torch-sdk ../public-api.cpp)
set_target_properties(torch-sdk PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_include_directories(torch-sdk PRIVATE ${TORCH_INCLUDE_DIRS})
separate_arguments(torch_compile_flags NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(torch-sdk PRIVATE -UNDEBUG ${torch_compile_flags})
target_link_libraries(torch-sdk PRIVATE "${TORCH_LIBRARIES}" Threads::Threads ${CMAKE_DL_LIBS} rt)
